function filesFor(workload) {
 if(workload==="google") return manifest.files;
 if(workload==="unicode-keys-nfc" || workload==="unicode-key-collisions") {
  const text = workload==="unicode-keys-nfc"
   ? JSON.stringify(Object.fromEntries(Array.from({length:2048},(_,i)=>[`café_${i}`,i])),null,2)+"\n"
   : JSON.stringify(Array.from({length:256},(_,i)=>({[`é_${i}`]:0,[`e\u0301_${i}`]:1})),null,2)+"\n";
  return [{path:workload+".json",bytes:Buffer.byteLength(text),sha256:sha256(text),syntheticText:text}];
 }
 return manifest.synthetic.find(s=>s.name===workload).files;
}
const bytesFor = (file) => file.syntheticText===undefined ? readFileSync(join(here,file.snapshot)) : Buffer.from(file.syntheticText);

 return paths.map(path=>[path,sha256(readFileSync(join(productRoot,path)))]);
}
